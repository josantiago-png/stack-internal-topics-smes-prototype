        @minUniquePostsActivity int = 5,
        @minBronzeScore int = 25,

with
sme_tags as (
    SELECT distinct um.SubjectMatterExpertTagId as TagId
    FROM   UserMappings um
    -- userId is the actual user, or is a user group that the user is a member of
    WHERE  (
            (um.UserId >= -1 AND um.UserId = @userId)
            OR
            (
             um.UserId < -1
             AND
             um.UserId IN (SELECT distinct UserGroupId FROM UserMappings WHERE UserId = @userId)
            )
           )
           and um.DeletionDate IS NULL
           and um.SubjectMatterExpertTagId IS NOT NULL
),
watched_ignored_tags as (
	SELECT t.Id as TagId, tsr.TagSetTypeId
	FROM   TagSetRows tsr
	JOIN   Tags t on tsr.TagName = t.[Name]
	WHERE  tsr.OwnerUserId = @userId
),
community_tags as (
	SELECT distinct ct.TagId
	FROM CommunityMemberships cm
	JOIN Communities c on cm.CommunityId = c.Id
	JOIN CommunityTags ct on c.Id = ct.CommunityId
	WHERE cm.UserId = @userId
      AND c.DeletionDate is null
),
badge_tags as (
	SELECT t.Id as TagId, b.Class
	FROM Users2Badges u2b
	JOIN Badges b on u2b.BadgeId = b.Id
	JOIN Tags t on t.[Name] = b.[Name]
	WHERE u2b.UserId = @userId
      AND b.TagBased = 1
),
/* Tags where user has significant progress towards a badge.
 * we are ok grabbing all progress here for any badge above
 * the minimum for bronze, even if this gives progress to badges
 * that have already been awarded, since the awarded badges
 * will be included higher up on the cascade anyway.
 * Also intentionally not checking to see if tag has enough questions have a badge:
 * Tag existence will be verified below, no need to complicate here unnecessarily.*/
badge_progress as (
	SELECT TagId, TotalAnswerCount, TotalAnswerScore
	FROM UserTagTotals
	WHERE UserId = @userId
	  AND TotalAnswerCount >= @minBronzeCount
	  AND TotalAnswerScore >= @minBronzeScore -- values for bronze
),
-- tags from distinct posts where user has commented min number of times
post_comments as (
	SELECT pt.TagId
	FROM (
		SELECT Distinct IsNull(p.ParentId, p.Id) as PostId -- questionId
		FROM PostComments pc
		LEFT OUTER JOIN Posts p on p.Id = pc.PostId
		WHERE pc.UserId = @userId AND pc.DeletionDate is null
	) as DistinctPostsCommented
	JOIN PostTags pt on pt.PostId = DistinctPostsCommented.PostId
    JOIN Posts p on DistinctPostsCommented.PostId = p.Id
    WHERE p.DeletionDate is null AND p.ClosedDate is null
	GROUP BY pt.TagId
	HAVING Count(DistinctPostsCommented.PostId) >= @minUniquePostsActivity
),
posts_edited as (
	SELECT pt.TagId
	FROM (
		SELECT Distinct IsNull(p.ParentId, p.Id) as PostId -- questionId
		FROM PostHistory ph
		LEFT OUTER JOIN Posts p on p.Id = ph.PostId
		WHERE ph.UserId = @userId
          AND ph.PostHistoryTypeId in (5, 4, 6)
	) as DistinctPostsEdited
	JOIN PostTags pt on pt.PostId = DistinctPostsEdited.PostId
	JOIN Posts p on p.Id = DistinctPostsEdited.PostId
    WHERE p.DeletionDate is null AND p.ClosedDate is null
	GROUP BY pt.TagId
	HAVING Count(DistinctPostsEdited.PostId) >= @minUniquePostsActivity
),
posts_voted as (
	SELECT pt.TagId
	FROM (
		SELECT Distinct IsNull(p.ParentId, p.Id) as PostId -- questionId
		FROM Posts2Votes pv
		LEFT OUTER JOIN Posts p on p.Id = pv.PostId
		WHERE pv.UserId = @userId
          AND pv.DeletionDate is null
          AND pv.VoteTypeId in (2, 3)
	) as DistinctPostsVoted
	JOIN PostTags pt on pt.PostId = DistinctPostsVoted.PostId
	JOIN Posts p on p.Id = DistinctPostsVoted.PostId
    WHERE p.DeletionDate is null AND p.ClosedDate is null
	GROUP BY pt.TagId
	HAVING Count(DistinctPostsVoted.PostId) >= @minUniquePostsActivity
),
all_data as (

    SELECT TagId,
           1 as TagPrefType
    FROM sme_tags

	UNION

	SELECT TagId,
           2 as TagPrefType
    FROM watched_ignored_tags
	WHERE  TagSetTypeId = 1

	UNION

	SELECT TagId,
		   3 as TagPrefType
	FROM   community_tags

	UNION

	SELECT TagId,
		   CASE WHEN Class = 1 THEN 4       -- BadgeClass.Gold
				WHEN Class = 2 THEN 6     -- BadgeClass.Silver
				ELSE 8 END AS TagPrefType -- BadgeClass.Bronze
	FROM badge_tags

	UNION

	-- if has above gold min progress, then include in gold tag pref
	-- otherwise try for silver, otherwise try for bronze
	SELECT TagId,
	  CASE WHEN TotalAnswerCount >= @minGoldCount and TotalAnswerScore >= @minGoldScore then 5 -- gold badge progress
		   WHEN TotalAnswerCount >= @minSilverCount and TotalAnswerScore >= @minSilverScore then 7 -- silver badge progress
		   ELSE 9 END as TagPrefType -- if it didnt have enough for gold or silver, then has to be bronze
	FROM badge_progress

	UNION

	SELECT TagId,
	       10 as TagPrefType
	From post_comments

	UNION

	SELECT TagId,
	       11 as TagPrefType
	From posts_edited

	UNION

	SELECT TagId,
	       12 as TagPrefType
	From posts_voted
)
-- Get the lowest pref type (lower is better) for each badge
-- to present for filtering on the server
Select t.[Name],
	   ad.TagId,
       TagPrefType as PrefType
FROM all_data ad
JOIN Tags t on t.Id = ad.TagId -- verifies the existence of the tag, as well as providing name
LEFT OUTER JOIN watched_ignored_tags it on t.Id = it.TagId and it.TagSetTypeId = 2
WHERE it.TagId IS NULL -- exclude ignored tags
    AND t.IsArticleRequired = 0 -- exclude article required tags;
